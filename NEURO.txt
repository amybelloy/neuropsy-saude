        <script>
            (function () {
                // ── Canvas de neurônios ───────────────────────────────────────────
                var canvas = document.getElementById('neuronCanvas');
                var ctx = canvas.getContext('2d');
                var W, H, particles = [], lines = [], mouse = { x: -9999, y: -9999, active: false };
                var PARTICLE_COUNT = 80;
                var MAX_DIST = 180;
                var colors = ['rgba(255,100,180,', 'rgba(180,100,255,', 'rgba(0,220,220,', 'rgba(255,220,80,'];

                function resize() {
                    W = canvas.width = canvas.offsetWidth;
                    H = canvas.height = canvas.offsetHeight;
                }

                function Particle() {
                    this.x = Math.random() * W;
                    this.y = Math.random() * H;
                    this.vx = (Math.random() - 0.5) * 0.4;
                    this.vy = (Math.random() - 0.5) * 0.4;
                    this.r = Math.random() * 2.5 + 1;
                    this.color = colors[Math.floor(Math.random() * colors.length)];
                    this.pulse = Math.random() * Math.PI * 2;
                    this.pulseSpeed = Math.random() * 0.03 + 0.01;
                    this.fired = false;
                    this.fireTimer = 0;
                }

                Particle.prototype.update = function () {
                    this.x += this.vx;
                    this.y += this.vy;
                    this.pulse += this.pulseSpeed;
                    if (this.x < 0 || this.x > W) this.vx *= -1;
                    if (this.y < 0 || this.y > H) this.vy *= -1;
                    // Reagir ao mouse
                    if (mouse.active) {
                        var dx = this.x - mouse.x, dy = this.y - mouse.y;
                        var d = Math.sqrt(dx * dx + dy * dy);
                        if (d < 120) {
                            var force = 0.6 * (1 - d / 120);
                            this.vx += dx / d * force;
                            this.vy += dy / d * force;
                            var spd = Math.sqrt(this.vx * this.vx + this.vy * this.vy);
                            if (spd > 3) { this.vx = this.vx / spd * 3; this.vy = this.vy / spd * 3; }
                        }
                    }
                };

                Particle.prototype.draw = function () {
                    var alpha = 0.5 + 0.5 * Math.sin(this.pulse);
                    var glow = this.r * (2 + 2 * Math.sin(this.pulse));
                    // Brilho externo
                    var grad = ctx.createRadialGradient(this.x, this.y, 0, this.x, this.y, glow * 3);
                    grad.addColorStop(0, this.color + '' + alpha + ')');
                    grad.addColorStop(1, this.color + '0)');
                    ctx.beginPath();
                    ctx.arc(this.x, this.y, glow * 3, 0, Math.PI * 2);
                    ctx.fillStyle = grad;
                    ctx.fill();
                    // Núcleo
                    ctx.beginPath();
                    ctx.arc(this.x, this.y, this.r, 0, Math.PI * 2);
                    ctx.fillStyle = this.color + '' + (alpha) + ')';
                    ctx.fill();
                };

                function initParticles() {
                    particles = [];
                    for (var i = 0; i < PARTICLE_COUNT; i++) particles.push(new Particle());
                }

                function drawConnections() {
                    for (var i = 0; i < particles.length; i++) {
                        for (var j = i + 1; j < particles.length; j++) {
                            var dx = particles[i].x - particles[j].x;
                            var dy = particles[i].y - particles[j].y;
                            var d = Math.sqrt(dx * dx + dy * dy);
                            var mDist = MAX_DIST, alpha = 0;
                            if (mouse.active) {
                                var mx = (particles[i].x + particles[j].x) / 2 - mouse.x;
                                var my = (particles[i].y + particles[j].y) / 2 - mouse.y;
                                var md = Math.sqrt(mx * mx + my * my);
                                if (md < 250) mDist = MAX_DIST + 100;
                            }
                            if (d < mDist) {
                                alpha = (1 - d / mDist) * 0.5;
                                if (mouse.active) {
                                    var mx2 = (particles[i].x + particles[j].x) / 2 - mouse.x;
                                    var my2 = (particles[i].y + particles[j].y) / 2 - mouse.y;
                                    var md2 = Math.sqrt(mx2 * mx2 + my2 * my2);
                                    if (md2 < 250) alpha *= (1.5 + 1 * (1 - md2 / 250));
                                }
                                var cx = (particles[i].x + particles[j].x) / 2 / W;
                                var r = Math.round(180 + cx * 75);
                                var g = Math.round(80 + (1 - cx) * 120);
                                var b = Math.round(255 - cx * 120);
                                ctx.beginPath();
                                ctx.strokeStyle = 'rgba(' + r + ',' + g + ',' + b + ',' + alpha + ')';
                                ctx.lineWidth = mouse.active ? 1.2 : 0.7;
                                ctx.moveTo(particles[i].x, particles[i].y);
                                ctx.lineTo(particles[j].x, particles[j].y);
                                ctx.stroke();
                            }
                        }
                    }
                }

                var signals = [];
                function fireSignal() {
                    if (!mouse.active) return;
                    var close = [], best = Infinity, src = null;
                    particles.forEach(function (p) {
                        var d = Math.hypot(p.x - mouse.x, p.y - mouse.y);
                        if (d < best) { best = d; src = p; }
                    });
                    if (!src) return;
                    var neighbors = particles.filter(function (p) {
                        return p !== src && Math.hypot(p.x - src.x, p.y - src.y) < MAX_DIST + 80;
                    });
                    if (neighbors.length === 0) return;
                    var target = neighbors[Math.floor(Math.random() * neighbors.length)];
                    signals.push({
                        x: src.x, y: src.y,
                        tx: target.x, ty: target.y,
                        progress: 0, speed: 0.035,
                        color: src.color
                    });
                }

                function updateSignals() {
                    signals = signals.filter(function (s) {
                        s.progress += s.speed;
                        s.x = s.x + (s.tx - s.x) * s.speed * 8;
                        s.y = s.y + (s.ty - s.y) * s.speed * 8;
                        var alpha = 1 - s.progress;
                        if (alpha <= 0) return false;
                        var grad = ctx.createRadialGradient(s.x, s.y, 0, s.x, s.y, 12);
                        grad.addColorStop(0, s.color + alpha + ')');
                        grad.addColorStop(1, s.color + '0)');
                        ctx.beginPath();
                        ctx.arc(s.x, s.y, 10, 0, Math.PI * 2);
                        ctx.fillStyle = grad;
                        ctx.fill();
                        return s.progress < 1;
                    });
                }

                var frameCount = 0;
                function animate() {
                    requestAnimationFrame(animate);
                    ctx.clearRect(0, 0, W, H);
                    drawConnections();
                    particles.forEach(function (p) { p.update(); p.draw(); });
                    updateSignals();
                    frameCount++;
                    if (mouse.active && frameCount % 8 === 0) fireSignal();
                }

                var section = document.getElementById('hero-cinema');
                if (section) {
                    section.addEventListener('mousemove', function (e) {
                        var r = section.getBoundingClientRect();
                        mouse.x = e.clientX - r.left;
                        mouse.y = e.clientY - r.top;
                        mouse.active = true;
                        var bl = document.getElementById('brainLeft');
                        var br = document.getElementById('brainRight');
                        var xN = (mouse.x / W - 0.5) * 2;
                        var yN = (mouse.y / H - 0.5) * 2;
                        if (bl) bl.style.transform = 'translateY(calc(-50% + ' + (yN * 10) + 'px)) translateX(' + (xN * 8) + 'px)';
                        if (br) br.style.transform = 'translateY(calc(-50% + ' + (yN * 10) + 'px)) translateX(' + (xN * -8) + 'px)';
                    });
                    section.addEventListener('mouseleave', function () {
                        mouse.active = false;
                        mouse.x = -9999; mouse.y = -9999;
                        var bl = document.getElementById('brainLeft');
                        var br = document.getElementById('brainRight');
                        if (bl) bl.style.transform = 'translateY(-50%)';
                        if (br) br.style.transform = 'translateY(-50%)';
                    });
                }

                window.addEventListener('resize', function () { resize(); initParticles(); });
                resize();
                initParticles();
                animate();
            })();
        </script>
    </section>

document.addEventListener('DOMContentLoaded', () => {
    // Menu responsivo
    const toggle = document.querySelector('.menu-toggle');
    const navLinks = document.querySelector('.nav-links');

    if(toggle && navLinks) {
        toggle.addEventListener('click', () => {
            navLinks.classList.toggle('active');
        });
    }

    // Header fixo muda ao rolar a página
    const header = document.querySelector('header');
    window.addEventListener('scroll', () => {
        if(window.scrollY > 50) {
            header.classList.add('scrolled');
        } else {
            header.classList.remove('scrolled');
        }
    });

    // Observer para animação reveladora de elementos (fade-up)
    const observerOptions = {
        root: null,
        rootMargin: '0px',
        threshold: 0.15
    };

    const observer = new IntersectionObserver((entries, observer) => {
        entries.forEach(entry => {
            if(entry.isIntersecting) {
                entry.target.classList.add('active');
                observer.unobserve(entry.target); // Anima apenas uma vez
            }
        });
    }, observerOptions);

    const revealElements = document.querySelectorAll('.reveal');
    revealElements.forEach(el => {
        observer.observe(el);
    });

    // FAQ Accordion
    const faqQuestions = document.querySelectorAll('.faq-question');
    faqQuestions.forEach(question => {
        question.addEventListener('click', () => {
            const answer = question.nextElementSibling;
            const isOpen = question.classList.contains('active');

            // Fecha outros abertos
            faqQuestions.forEach(q => {
                q.classList.remove('active');
                q.nextElementSibling.style.maxHeight = null;
            });

            // Se não estava aberto, abre o clicado
            if (!isOpen) {
                question.classList.add('active');
                answer.style.maxHeight = answer.scrollHeight + "px";
            }
        });
    });
});


class Particle {
    constructor(canvas, ctx) {
        this.canvas = canvas;
        this.ctx = ctx;
        this.reset();
        
        // Espalhamento inicial pelo canvas
        this.x = Math.random() * canvas.width;
        this.y = Math.random() * canvas.width;
    }

    reset() {
        // Posição será influenciada pelas forças, não reseta ao clicar
        this.size = Math.random() * 2.5 + 1.5; // Bolinhas pequenas
        
        // Tons pasteis (azul, lilás, verde suave, rosa suave, laranja suave, amarelo claro)
        const colors = [
            'rgba(163, 196, 243, 0.7)', // azul
            'rgba(209, 196, 233, 0.7)', // lilás
            'rgba(161, 193, 166, 0.7)', // verde
            'rgba(248, 187, 208, 0.7)', // rosa
            'rgba(255, 204, 128, 0.7)', // laranja
            'rgba(255, 249, 196, 0.7)'  // amarelo
        ];
        this.color = colors[Math.floor(Math.random() * colors.length)];
        
        this.vx = (Math.random() - 0.5) * 1;
        this.vy = (Math.random() - 0.5) * 1;
        
        // Variabilidade do comportamento (personalidade da bolinha)
        this.friction = Math.random() * 0.04 + 0.92;
        this.mouseInfluence = Math.random() * 0.003 + 0.001; // Efeito reduzido e mais lento 
        this.wander = Math.random() * 0.5 + 0.2; // Desvio próprio para organicidade
        this.angle = Math.random() * Math.PI * 2;
    }

    update(mouseX, mouseY) {
        // Se houver mouse, atrai levemente
        if (mouseX && mouseY) {
            const dx = mouseX - this.x;
            const dy = mouseY - this.y;
            
            // Força de atração suave
            this.vx += dx * this.mouseInfluence;
            this.vy += dy * this.mouseInfluence;
        }

        // Movimento orgânico extra (flutuação)
        this.angle += 0.05;
        this.vx += Math.cos(this.angle) * this.wander * 0.1;
        this.vy += Math.sin(this.angle) * this.wander * 0.1;

        // Atrito para não acelerar infinitamente
        this.vx *= this.friction;
        this.vy *= this.friction;

        // Atualiza posição
        this.x += this.vx;
        this.y += this.vy;

        // Evita aglomeração com um limitador de velocidade extrema
        const speed = Math.sqrt(this.vx * this.vx + this.vy * this.vy);
        const maxSpeed = 1.8; // Velocidade máxima muito menor
        if (speed > maxSpeed) {
            this.vx = (this.vx / speed) * maxSpeed;
            this.vy = (this.vy / speed) * maxSpeed;
        }
    }

    draw() {
        this.ctx.beginPath();
        this.ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        
        // Gradiente radial para efeito premium
        const grad = this.ctx.createRadialGradient(
            this.x - this.size/4, 
            this.y - this.size/4, 
            0, 
            this.x, 
            this.y, 
            this.size
        );
        grad.addColorStop(0, '#ffffff'); // Centro iluminado
        grad.addColorStop(1, this.color); // Borda colorida

        this.ctx.fillStyle = grad;
        this.ctx.fill();
        this.ctx.closePath();
    }
}

class ParticleSystem {
    constructor() {
        this.canvas = document.createElement('canvas');
        this.ctx = this.canvas.getContext('2d');
        
        this.particles = [];
        this.mouseX = null;
        this.mouseY = null;
        
        this.initCanvas();
        this.initParticles();
        this.addEventListeners();
        
        this.animate();
    }

    initCanvas() {
        this.canvas.style.position = 'fixed';
        this.canvas.style.top = '0';
        this.canvas.style.left = '0';
        this.canvas.style.width = '100vw';
        this.canvas.style.height = '100vh';
        this.canvas.style.pointerEvents = 'none';
        this.canvas.style.zIndex = '-999'; // Fundo total
        document.body.appendChild(this.canvas);
        
        this.resize();
    }

    resize() {
        this.canvas.width = window.innerWidth;
        this.canvas.height = window.innerHeight;
    }

    initParticles() {
        // Reduz quantidade em celulares para performance
        const particleCount = window.innerWidth > 768 ? 70 : 30;
        this.particles = [];
        for (let i = 0; i < particleCount; i++) {
            this.particles.push(new Particle(this.canvas, this.ctx));
        }
    }

    addEventListeners() {
        window.addEventListener('resize', () => {
            this.resize();
            this.initParticles(); // Reinicia com contagem certa para tela
        });

        document.addEventListener('mousemove', (e) => {
            this.mouseX = e.clientX;
            this.mouseY = e.clientY;
        });

        // Solta as bolinhas quando o mouse sai da tela
        document.addEventListener('mouseleave', () => {
            this.mouseX = null;
            this.mouseY = null;
        });
    }

    applySeparation() {
        // Lógica de "cardume" - Separação
        for (let i = 0; i < this.particles.length; i++) {
            for (let j = i + 1; j < this.particles.length; j++) {
                const p1 = this.particles[i];
                const p2 = this.particles[j];
                
                const dx = p1.x - p2.x;
                const dy = p1.y - p2.y;
                const dist = Math.sqrt(dx * dx + dy * dy);
                
                const minDist = 35; // Distância que elas tentam manter
                
                if (dist < minDist && dist > 0) {
                    const force = (minDist - dist) / minDist;
                    p1.vx += (dx / dist) * force * 0.5;
                    p1.vy += (dy / dist) * force * 0.5;
                    p2.vx -= (dx / dist) * force * 0.5;
                    p2.vy -= (dy / dist) * force * 0.5;
                }
            }
        }
    }

    animate() {
        // Fundo transparente com leve rastro (trail blur effect)
        this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

        this.applySeparation();

        this.particles.forEach(p => {
            p.update(this.mouseX, this.mouseY);
            // Rebater nas bordas suavemente se mouseX não existir
            if (!this.mouseX) {
               if (p.x < 0 || p.x > this.canvas.width) p.vx *= -1;
               if (p.y < 0 || p.y > this.canvas.height) p.vy *= -1;
            }
            p.draw();
        });

        requestAnimationFrame(() => this.animate());
    }
}

// Inicializa quando o DOM carregar
document.addEventListener('DOMContentLoaded', () => {
    new ParticleSystem();
});
